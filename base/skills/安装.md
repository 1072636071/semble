## 安装方式

将源目录下每个子目录（即每个技能）复制到目标目录下，覆盖同名目录。**engineering 全部安装**（`jxx-*` 通配 + `impeccable`，`jxx-agent-generator` 默认不安装）；**productivity 仅安装默认列表**（`jxx-teach`、`jxx-wait-what`、`skill-creator`、`skill-reviewer`、`skill-tester` 除外）。脚本同时**删除目标中已从源移除或已转为默认不安装的残留技能**（如 `jxx-writing-great-skills`），保证新旧技能同步。

```powershell
$srcEng = "E:\work\sp\JwikisSkills\base\skills\engineering"
$srcProd = "E:\work\sp\JwikisSkills\base\skills\productivity"

# engineering 默认不安装列表——需要手动安装的技能在此追加
$engExclude = @(
    "jxx-agent-generator"
)

# engineering 非 jxx-* 系列技能（同样默认安装）——新增时在此追加
$engExtra = @(
    "impeccable"
)

# productivity 默认安装列表——新增默认技能时在此追加
$prodSkills = @(
    "jxx-grill-me"
    "jxx-grilling"
    "jxx-handoff"
    "jxx-to-questionnaire"
    "jxx-writing-for-agents"
)

$targets = @(
    "C:\Users\jxc123\.codebuddy\skills"
    "C:\Users\jxc123\.agents\skills"
    "C:\Users\jxc123\.codeartsdoer\skills"
    "C:\Users\jxc123\.qoder-cn\skills"
    "C:\Users\jxc123\.trae-cn\skills"
    "C:\Users\jxc123\.workbuddy\skills"
    "C:\Users\jxc123\.qoderworkcn\skills"
)

foreach ($dest in $targets) {
    # 1) engineering：复制全部 jxx-* 技能（默认不安装的除外）+ 非 jxx-* 技能（impeccable）
    Get-ChildItem "$srcEng\jxx-*" -Directory | ForEach-Object {
        if ($engExclude -notcontains $_.Name) {
            Copy-Item -Path $_.FullName -Destination $dest -Recurse -Force
        }
    }
    foreach ($extra in $engExtra) {
        $src = "$srcEng\$extra"
        if (Test-Path $src) { Copy-Item -Path $src -Destination $dest -Recurse -Force }
    }
    # 2) productivity：复制默认安装列表
    foreach ($skill in $prodSkills) {
        $src = "$srcProd\$skill"
        if (Test-Path $src) { Copy-Item -Path $src -Destination $dest -Recurse -Force }
    }
    # 3) 清理残留：目标中存在、但两个源目录都不再有、或被列为默认不安装的 jxx-* 技能目录
    #    （同时检查 engineering 与 productivity 源，避免误删源中仍存在的技能）
    Get-ChildItem $dest -Directory -Filter "jxx-*" | ForEach-Object {
        $inEng = Test-Path "$srcEng\$($_.Name)"
        $inProd = Test-Path "$srcProd\$($_.Name)"
        if ((-not $inEng -and -not $inProd) -or $engExclude -contains $_.Name) {
            Remove-Item $_.FullName -Recurse -Force
            Write-Host "  cleaned: $($_.Name)"
        }
    }
}
```

## 验证

安装完成后，对所有平台验证本次修改过的技能是否部署正确：

```powershell
$srcEng = "E:\work\sp\JwikisSkills\base\skills\engineering"
$srcProd = "E:\work\sp\JwikisSkills\base\skills\productivity"

# 本次新增/修改的技能（根据 git diff 维护）
$modifiedSkills = @(
    @{ Name = "jxx-wizard";            Src = $srcEng }
    @{ Name = "jxx-ask-matt";          Src = $srcEng }
    @{ Name = "jxx-diagnosing-bugs";   Src = $srcEng }
    @{ Name = "jxx-prototype";         Src = $srcEng }
    @{ Name = "jxx-code-review";       Src = $srcEng }
    @{ Name = "jxx-codebase-design";   Src = $srcEng }
    @{ Name = "jxx-improve-codebase-architecture"; Src = $srcEng }
    @{ Name = "jxx-grilling";          Src = $srcProd }
    @{ Name = "jxx-to-questionnaire";  Src = $srcProd }
    @{ Name = "jxx-writing-for-agents";Src = $srcProd }
)

$targets = @(
    "C:\Users\jxc123\.codebuddy\skills"
    "C:\Users\jxc123\.agents\skills"
    "C:\Users\jxc123\.codeartsdoer\skills"
    "C:\Users\jxc123\.qoder-cn\skills"
    "C:\Users\jxc123\.trae-cn\skills"
    "C:\Users\jxc123\.workbuddy\skills"
    "C:\Users\jxc123\.qoderworkcn\skills"
)

foreach ($dest in $targets) {
    $label = Split-Path $dest -Parent | Split-Path -Leaf
    Write-Host "=== $label ==="
    foreach ($skill in $modifiedSkills) {
        $srcFile = "$($skill.Src)\$($skill.Name)\SKILL.md"
        $dstFile = "$dest\$($skill.Name)\SKILL.md"
        if (Test-Path $dstFile) {
            $srcHash = (Get-FileHash $srcFile -Algorithm SHA256).Hash
            $dstHash = (Get-FileHash $dstFile -Algorithm SHA256).Hash
            if ($srcHash -eq $dstHash) { Write-Host "  $($skill.Name) : OK" }
            else { Write-Host "  $($skill.Name) : MISMATCH" }
        } else {
            Write-Host "  $($skill.Name) : MISSING"
        }
    }
}
```

若 Copy-Item 报 `untrusted mount point` 错误，改用 robocopy：

```powershell
$destBase = "C:\Users\jxc123\.agents\skills"
$srcEng = "E:\work\sp\JwikisSkills\base\skills\engineering"
$srcProd = "E:\work\sp\JwikisSkills\base\skills\productivity"

# engineering（默认不安装的除外 + 非 jxx-* 技能）
$engExclude = @("jxx-agent-generator")
Get-ChildItem "$srcEng\jxx-*" -Directory | ForEach-Object {
    if ($engExclude -notcontains $_.Name) {
        robocopy $_.FullName "$destBase\$($_.Name)" /E /IS /IT /NFL /NDL /NJH /NJS
    }
}
robocopy "$srcEng\impeccable" "$destBase\impeccable" /E /IS /IT /NFL /NDL /NJH /NJS
# productivity
@("jxx-grill-me", "jxx-grilling", "jxx-handoff", "jxx-to-questionnaire", "jxx-writing-for-agents") | ForEach-Object {
    robocopy "$srcProd\$_" "$destBase\$_" /E /IS /IT /NFL /NDL /NJH /NJS
}
```

## 注意

- 不要复制 `README.md` 和 `正常的工作流.md`（非技能目录）
- 安装后重启各平台或重新加载技能列表
- 源目录中的 `.md`、`.json`、`.txt`、`.py`、`.sh` 等附属文件一并复制，技能可能依赖
- 每次修改技能源文件后，执行安装 + 验证，确保所有平台同步
- 若某技能已从源删除，安装脚本会清理目标中的残留目录；如需保留旧版，手动从目标移出
